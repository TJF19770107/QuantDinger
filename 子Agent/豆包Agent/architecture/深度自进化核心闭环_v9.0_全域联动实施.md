## 四、Obsidian 全域联动 v2.0

### 4.1 双向同步机制

```python
# obsidian_sync_bridge_v2.py
"""
Obsidian知识图谱 ←→ 豆包Agent 双向同步桥接 v2.0
R54升级: 从单向产出入库升级为双向实时同步
"""
import json
import hashlib
from pathlib import Path
from datetime import datetime

OBSIDIAN_VAULT = Path(r"E:\龙虾AI主控中心\Obsidian")
AGENT_STATE = Path(r"E:\龙虾AI主控中心\我的AI分身\子Agent\豆包Agent")

class ObsidianSyncBridge:
    """Obsidian ↔ Agent 双向同步桥接"""
    
    def __init__(self):
        self.sync_registry = Path(OBSIDIAN_VAULT / ".sync_registry.json")
        self._load_registry()
    
    # ========== Agent → Obsidian ==========
    
    def push_skill_update(self, skill: dict):
        """新技能 → Obsidian知识图谱节点"""
        note_path = OBSIDIAN_VAULT / "技能图谱" / f"{skill['name']}.md"
        content = self._skill_to_markdown(skill)
        note_path.parent.mkdir(parents=True, exist_ok=True)
        note_path.write_text(content, encoding='utf-8')
        self._update_registry('skill', skill['skill_id'], note_path)
    
    def push_capability_matrix(self, matrix: dict):
        """26维能力矩阵 → Obsidian看板"""
        note_path = OBSIDIAN_VAULT / "能力看板" / "26维能力矩阵.md"
        content = self._matrix_to_dataview(matrix)
        note_path.parent.mkdir(parents=True, exist_ok=True)
        note_path.write_text(content, encoding='utf-8')
    
    def push_safety_event(self, event: dict):
        """MCP安全事件 → Obsidian安全日志"""
        note_path = OBSIDIAN_VAULT / "安全日志" / f"{event['date']}_{event['type']}.md"
        content = f"""---
type: mcp_safety_event
attack_type: {event['type']}
antibody_deployed: {event.get('antibody', 'none')}
nrp_impact: {event.get('nrp_delta', 0)}
timestamp: {event['timestamp']}
---

# MCP安全事件: {event['type']}

- **攻击向量**: {event['vector']}
- **检测方式**: {event['detection_method']}
- **防御动作**: {event['defense_action']}
- **NRP变化**: {event.get('nrp_delta', 0):+.2f}
- **关联抗体**: [[抗体/{event.get('antibody', 'none')}]]
"""
        note_path.parent.mkdir(parents=True, exist_ok=True)
        note_path.write_text(content, encoding='utf-8')
    
    def push_protocol_update(self, protocol: dict):
        """新协议 → Obsidian协议图谱"""
        note_path = OBSIDIAN_VAULT / "协议图谱" / f"协议{protocol['id']}.md"
        content = f"""---
protocol_id: {protocol['id']}
version: {protocol['version']}
status: {protocol['status']}
dimension: {protocol['dimension']}
created: {protocol['created']}
---
# 协议 #{protocol['id']}: {protocol['name']}

{protocol['content']}

## 关联
- 技能: {', '.join(protocol.get('skills', []))}
- 维度: {protocol['dimension']}
"""
        note_path.parent.mkdir(parents=True, exist_ok=True)
        note_path.write_text(content, encoding='utf-8')
    
    # ========== Obsidian → Agent ==========
    
    def pull_user_feedback(self) -> list[dict]:
        """从Obsidian读取用户标注的反馈/缺口"""
        feedbacks = []
        feedback_dir = OBSIDIAN_VAULT / "反馈与缺口"
        if feedback_dir.exists():
            for note in feedback_dir.glob("*.md"):
                content = note.read_text(encoding='utf-8')
                # 解析YAML front matter
                feedbacks.append(self._parse_feedback(content, note))
        return feedbacks
    
    def pull_manual_skills(self) -> list[dict]:
        """从Obsidian读取手动编写的技能"""
        skills = []
        manual_dir = OBSIDIAN_VAULT / "手动技能"
        if manual_dir.exists():
            for note in manual_dir.glob("*.md"):
                skills.append(self._parse_skill(content := note.read_text(encoding='utf-8')))
        return skills
    
    def sync_loop(self):
        """双向同步主循环"""
        # Agent → Obsidian
        self._push_all_changes()
        # Obsidian → Agent (反馈驱动进化)
        feedbacks = self.pull_user_feedback()
        if feedbacks:
            self._process_feedback_for_evolution(feedbacks)
    
    # ========== 内部方法 ==========
    
    def _skill_to_markdown(self, skill: dict) -> str:
        lines = [
            "---",
            f"skill_id: {skill['skill_id']}",
            f"category: {skill['category']}",
            f"version: {skill.get('version', '1.0')}",
            f"success_rate: {skill.get('quality_metrics', {}).get('success_rate', 0)}",
            "---",
            f"# {skill['name']}",
            f"\n{skill['description']}\n",
            "## 触发词",
            ', '.join(skill.get('trigger_words', [])),
            "\n## 工具链",
            ' → '.join(skill.get('tools_chain', [])),
            "\n## 关联",
        ]
        # 关联其他Obsidian笔记
        for tool in skill.get('tools_chain', [])[:3]:
            lines.append(f"- [[工具/{tool}]]")
        lines.append(f"- [[维度/{skill.get('category', 'general')}]]")
        return '\n'.join(lines)
    
    def _matrix_to_dataview(self, matrix: dict) -> str:
        lines = [
            "# 26维能力矩阵",
            "",
            "```dataview",
            "TABLE score, target, gap, trend",
            "FROM \"能力看板\"",
            "WHERE type = \"dimension\"",
            "SORT gap DESC",
            "```",
            "",
            "## 当前评分",
            ""
        ]
        for dim, data in matrix.items():
            gap = data.get('target', 100) - data.get('score', 0)
            trend = data.get('trend', '→')
            lines.append(f"- **{dim}**: {data.get('score', 0)}/100 (目标{data.get('target', 100)}, 缺口{gap}) {trend}")
        return '\n'.join(lines)
    
    def _load_registry(self):
        if self.sync_registry.exists():
            self.registry = json.loads(self.sync_registry.read_text())
        else:
            self.registry = {'skills': {}, 'protocols': {}, 'events': {}}
    
    def _update_registry(self, category: str, item_id: str, path: Path):
        self.registry.setdefault(category, {})[item_id] = {
            'path': str(path),
            'last_sync': datetime.now().isoformat(),
            'hash': hashlib.md5(path.read_bytes()).hexdigest()
        }
        self.sync_registry.write_text(json.dumps(self.registry, indent=2))
```

### 5.1 界面感知引擎

```python
# ai_on_ui_bridge.py
"""
AI on UI 交互桥接 v1.0
R54新增: 打通UI感知→Agent推理→UI操作闭环
"""
import time
from dataclasses import dataclass
from typing import Optional

@dataclass
class UIState:
    """当前UI状态快照"""
    active_window: str         # 活动窗口标题
    active_app: str            # 活动应用名
    cursor_position: tuple     # 光标位置
    selected_text: str         # 选中文本
    clipboard_content: str     # 剪贴板内容
    visible_elements: list     # 可见UI元素
    screenshot_base64: str     # 屏幕截图

class AIOnUIBridge:
    """AI on UI: 感知→理解→操作 闭环"""
    
    def __init__(self):
        self.state_history = deque(maxlen=100)
        self.action_history = deque(maxlen=500)
    
    # ====== 感知层 ======
    
    def capture_ui_state(self) -> UIState:
        """捕获当前UI状态"""
        return UIState(
            active_window=self._get_active_window(),
            active_app=self._get_active_app(),
            cursor_position=self._get_cursor_position(),
            selected_text=self._get_selected_text(),
            clipboard_content=self._get_clipboard(),
            visible_elements=self._enumerate_ui_elements(),
            screenshot_base64=self._capture_screenshot()
        )
    
    # ====== 理解层 ======
    
    def understand_context(self, state: UIState, user_intent: str) -> dict:
        """理解UI上下文：用户在做什么、需要什么"""
        context = {
            'user_is_editing': 'editor' in state.active_app.lower(),
            'user_is_browsing': 'browser' in state.active_app.lower(),
            'user_is_coding': state.active_app.lower() in ['vscode', 'pycharm', 'cursor'],
            'selected_content_type': self._classify_content(state.selected_text),
            'clipboard_content_type': self._classify_content(state.clipboard_content),
            'suggested_actions': self._infer_actions(state, user_intent)
        }
        return context
    
    # ====== 操作层 ======
    
    def execute_ui_action(self, action: dict) -> bool:
        """执行UI操作（通过桌面控制Agent）"""
        action_type = action['type']
        
        dispatchers = {
            'type_text': self._type_text,
            'click_element': self._click_element,
            'scroll_to': self._scroll_to,
            'select_text': self._select_text,
            'open_app': self._open_application,
            'fill_form': self._fill_form,
            'take_screenshot': self._capture_screenshot,
        }
        
        if action_type in dispatchers:
            result = dispatchers[action_type](action['params'])
            self.action_history.append({
                'action': action,
                'result': result,
                'timestamp': time.time()
            })
            return result
        
        return False
    
    # ====== 进化联动 ======
    
    def report_to_sica(self):
        """向SICA上报UI交互数据，驱动自进化"""
        recent_actions = list(self.action_history)[-100:]
        
        # 分析高频操作模式
        patterns = self._analyze_ui_patterns(recent_actions)
        
        # 生成UI操作技能候选
        for pattern in patterns:
            if pattern.frequency >= 5 and pattern.success_rate > 0.8:
                # 触发技能自动萃取
                yield {
                    'source': 'ai_on_ui',
                    'pattern': pattern,
                    'suggested_skill': self._pattern_to_skill_candidate(pattern)
                }
```

### 6.3 全域联动事件总线

```python
# event_bus_universal.py
"""
全域联动事件总线 v1.0
统一汇聚 Obsidian / AI-on-UI / Desktop / SICA 事件
"""
from collections import deque
from enum import Enum

class EventSource(Enum):
    OBSIDIAN = "obsidian"
    AI_ON_UI = "ai_on_ui"
    DESKTOP = "desktop"
    SICA = "sica"
    CLAUDE_REASONER = "claude_reasoner"
    WORKFLOW_ENGINE = "workflow_engine"

class UniversalEventBus:
    """全域事件总线"""
    
    def __init__(self):
        self.subscribers: dict[EventSource, list[callable]] = {}
        self.event_log = deque(maxlen=10000)
    
    def subscribe(self, source: EventSource, handler: callable):
        self.subscribers.setdefault(source, []).append(handler)
    
    def publish(self, event: dict):
        """发布事件到所有订阅者"""
        source = EventSource(event.get('source', 'unknown'))
        
        # 记录事件
        self.event_log.append({
            **event,
            'timestamp': datetime.now().isoformat(),
            'event_id': hashlib.md5(str(event).encode()).hexdigest()[:12]
        })
        
        # 通知订阅者
        for handler in self.subscribers.get(source, []):
            try:
                handler(event)
            except Exception as e:
                logger.error(f"事件处理失败: {e}")
        
        # 跨源桥接：关键事件同步到SICA
        if event.get('severity') in ['high', 'critical']:
            self._bridge_to_sica(event)
    
    def _bridge_to_sica(self, event: dict):
        """事件桥接到SICA自进化"""
        sica_event = {
            'source': 'universal_event_bus',
            'original_event': event,
            'action': 'trigger_scan' if event['severity'] == 'critical' else 'enqueue'
        }
        self.publish({**sica_event, 'source': 'sica'})

# 全局单例
event_bus = UniversalEventBus()
```
