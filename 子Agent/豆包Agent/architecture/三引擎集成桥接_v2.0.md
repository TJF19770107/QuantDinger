## 四、统一编排入口

```python
class DoubaoAgentOrchestratorV2:
    """三引擎统一编排器 v2.0 (R23)"""
    
    def __init__(self):
        self.reasoning = ClaudeReasoningV6()
        self.workflow = VisualWorkflowV6()
        self.evolution = SelfEvolutionV7()
        self.event_bus = EventBusV2(events=12)
        self.obsidian_sync = ObsidianSyncEngine()
    
    def execute_full_cycle(self, user_input: str) -> FullCycleResult:
        """完整推理→编排→进化→同步闭环"""
        # Phase 1: 推理
        plan = self.reasoning.analyze(user_input)
        if plan.effort == EffortLevel.LOW:
            return self._fast_path(plan)
        
        # Phase 2: 编排
        dag = self.workflow.build_dag(plan)
        exec_result = self.workflow.execute(dag)
        self.event_bus.emit("EVT_001", plan)
        
        # Phase 3: 进化检查
        if self.evolution.should_evolve(exec_result):
            evo_result = self.evolution.evolve(exec_result)
            self.event_bus.emit("EVT_009", evo_result)
        
        # Phase 4: 快照保存
        snapshot = self.evolution.snapshot_manager.save(exec_result.trace)
        self.event_bus.emit("EVT_010", snapshot)
        
        # Phase 5: Obsidian同步
        self.obsidian_sync.sync_all_changed_files()
        
        return FullCycleResult(plan, exec_result, snapshot)
```
