```python
class SkillExtractor:
    """技能自动萃取引擎 - GenericAgent风格"""
    
    def __init__(self):
        self.task_clusterer = TaskClusterer()
        self.pattern_recognizer = PatternRecognizer()
        self.skill_validator = SkillValidator()
        self.skill_registry = SkillRegistry()
        self.min_success_samples = 3  # 最少成功样本数才触发萃取
    
    def extract_from_logs(self, trace_ids: List[str]) -> List[Skill]:
        """从执行日志中自动萃取技能"""
        
        # Phase 1: 收集成功执行轨迹
        successful_traces = []
        for tid in trace_ids:
            trace = LogBinder.query_by_trace(tid)
            if self._is_successful(trace):
                successful_traces.append(trace)
        
        # Phase 2: 按任务类型聚类
        clusters = self.task_clusterer.cluster(successful_traces)
        
        new_skills = []
        for cluster in clusters:
            if len(cluster) < self.min_success_samples:
                continue
            
            # Phase 3: 模式识别
            pattern = self.pattern_recognizer.extract(cluster)
            
            # Phase 4: 参数化抽象
            skill_template = self._abstract_parameters(pattern)
            
            # Phase 5: 技能验证（回测）
            if self.skill_validator.validate(skill_template, cluster):
                # Phase 6: 标准化入库
                skill = self._generate_skill(skill_template)
                self.skill_registry.register(skill)
                new_skills.append(skill)
        
        return new_skills
    
    def _abstract_parameters(self, pattern: ExecutionPattern) -> SkillTemplate:
        """将执行模式参数化抽象"""
        template = SkillTemplate(
            task_type=pattern.task_type,
            domain=pattern.domain,
            steps=[],
            variable_slots={},
            tool_chain=pattern.tool_chain
        )
        
        # 识别可参数化的部分
        for step in pattern.steps:
            params = {}
            for k, v in step.params.items():
                if self._is_variable(v, pattern.all_params):
                    slot_name = f"{{{{{k}}}}}"  # 模板变量
                    template.variable_slots[slot_name] = {
                        "type": type(v).__name__,
                        "examples": self._get_examples(k, pattern)
                    }
                    params[k] = slot_name
                else:
                    params[k] = v
            template.steps.append(StepTemplate(
                operation=step.operation,
                params=params,
                expected_output=step.expected_output
            ))
        
        return template
    
    def _generate_skill(self, template: SkillTemplate) -> Skill:
        """生成标准化Skill文件"""
        skill_content = f"""# {template.task_type} Skill
> 自动萃取于: {datetime.now().isoformat()}
> 来源样本: {template.sample_count}条成功执行
> 置信度: {template.confidence:.2f}

## 任务描述
{template.description}

## 执行步骤
"""
        for i, step in enumerate(template.steps, 1):
            skill_content += f"\n### Step {i}: {step.operation}\n"
            skill_content += f"- 参数: {step.params}\n"
            skill_content += f"- 预期产出: {step.expected_output}\n"
        
        skill_content += f"""
## 变量说明
{template.variable_slots}

## 适用条件
- 领域: {template.domain}
- 前置技能: {template.prerequisites}
- 调用频率: {template.avg_frequency}

## 性能指标
- 平均耗时: {template.avg_duration:.2f}s
- 成功率: {template.success_rate:.1%}
- 平均Token: {template.avg_tokens}
"""
        
        return Skill(
            name=template.task_type.replace(" ", "_").lower(),
            content=skill_content,
            template=template,
            confidence=template.confidence,
            extracted_at=datetime.now().isoformat()
        )
```

### 3.3 任务聚类算法

```python
class TaskClusterer:
    """基于多维特征的任务聚类器"""
    
    FEATURES = ["domain", "tool_chain", "complexity", "entity_types", "output_type"]
    
    def cluster(self, traces: List[ExecutionTrace]) -> List[List[ExecutionTrace]]:
        """将相似任务聚类"""
        vectors = [self._vectorize(t) for t in traces]
        
        # 使用DBSCAN的语义相似度变体
        clusters = []
        visited = set()
        
        for i, v in enumerate(vectors):
            if i in visited:
                continue
            
            cluster = [traces[i]]
            visited.add(i)
            
            for j, w in enumerate(vectors[i+1:], i+1):
                if j not in visited:
                    sim = self._similarity(v, w)
                    if sim > 0.75:  # 相似度阈值
                        cluster.append(traces[j])
                        visited.add(j)
            
            if len(cluster) >= 2:
                clusters.append(cluster)
        
        return clusters
    
    def _vectorize(self, trace: ExecutionTrace) -> np.ndarray:
        """将执行轨迹向量化"""
        vec = []
        # 领域编码
        vec.extend(self._onehot(trace.domain, self.domain_vocab))
        # 工具链编码
        vec.extend(self._toolchain_hash(trace.tool_chain))
        # 复杂度
        vec.append(trace.complexity_score)
        # 实体类型
        vec.extend(self._onehot_multi(trace.entity_types, self.entity_vocab))
        return np.array(vec)
```

class SICAReinforced:
    """SICA四步循环加固版 v3.0"""
    
    def __init__(self):
        self.checkpoints = {}
        self.snapshot_manager = SnapshotManager()
        self.gepa = GEPAEvaluator()
    
    def run_cycle(self, trigger: Trigger) -> CycleResult:
        """执行加固版SICA循环"""
        cycle_id = f"sica_{uuid4().hex[:8]}"
        trace = CycleTrace(cycle_id)
        
        try:
            # Step 1: Sense - 多源感知 + 校验
            sense_result = self._sense(trigger)
            if sense_result.confidence < 0.7:
                return CycleResult(success=False, reason="感知置信度不足")
            self.checkpoints["S1"] = Checkpoint("S1", sense_result)
            trace.add_step("S1", sense_result)
            
            # Step 2: Interpret - 因果推理 + 历史检索
            interpret_result = self._interpret(sense_result)
            if not interpret_result.root_cause:
                return CycleResult(success=False, reason="无法定位根因")
            self.checkpoints["S2"] = Checkpoint("S2", interpret_result)
            trace.add_step("S2", interpret_result)
            
            # Step 3: Compose - 多方案生成 + 择优
            candidates = self._compose_candidates(interpret_result)
            if len(candidates) < 2:
                # 补充分方案
                candidates.append(self._generate_alternative(interpret_result))
            
            best_plan = self._select_best(candidates)
            self.checkpoints["S3"] = Checkpoint("S3", best_plan)
            trace.add_step("S3", best_plan)
            
            # Step 4: Act - 原子化执行 + 偏差检测
            act_result = self._act_atomic(best_plan, trace)
            if act_result.deviation > 0.3:
                # 偏差过大，回溯到S3重新选方案
                trace.add_rollback("S4→S3", f"偏差{act_result.deviation:.2f}")
                candidates.remove(best_plan)
                if candidates:
                    best_plan = self._select_best(candidates)
                    act_result = self._act_atomic(best_plan, trace)
                else:
                    return CycleResult(success=False, reason="所有候选方案均失败")
            
            # 完成 → 保存版本快照
            self.snapshot_manager.save(trace)
            
            # GEPA后评估
            post_score = self.gepa.evaluate(trace)
            cycle_result = CycleResult(
                success=True,
                cycle_id=cycle_id,
                score=post_score,
                trace=trace
            )
            
            # 触发技能萃取检查
            self._maybe_extract_skill(trace)
            
            return cycle_result
            
        except Exception as e:
            # 异常自动回滚到最近检查点
            rollback_point = self._find_rollback_point()
            self.snapshot_manager.rollback(rollback_point)
            trace.add_rollback(f"exception→{rollback_point}", str(e))
            return CycleResult(success=False, reason=str(e), trace=trace)
    
    def _find_rollback_point(self) -> str:
        """找到最近的可用检查点"""
        for point in ["S3", "S2", "S1"]:
            if point in self.checkpoints:
                return point
        return "PRE_CYCLE"  # 回到循环前状态
```

@dataclass
class VersionSnapshot:
    """版本快照 - Git语义风格"""
    version: str            # v7.0.xxx
    parent: str             # 父版本
    timestamp: str          # ISO 8601
    state_hash: str         # 系统状态哈希
    files_changed: List[str]
    gepa_score: GEPAScore
    new_skills: List[str]   # 新增技能
    new_protocols: List[str] # 新增协议
    metrics: SystemMetrics  # 性能指标
    rollback_available: bool

class SnapshotManager:
    """版本快照管理器"""
    
    def __init__(self, snapshot_dir: str):
        self.snapshot_dir = snapshot_dir
        self.current_version = self._load_current_version()
    
    def save(self, trace: CycleTrace) -> VersionSnapshot:
        """保存当前版本快照"""
        version = self._next_version()
        state_hash = self._hash_system_state()
        
        snapshot = VersionSnapshot(
            version=version,
            parent=self.current_version,
            timestamp=datetime.now().isoformat(),
            state_hash=state_hash,
            files_changed=trace.files_modified,
            gepa_score=trace.gepa_score,
            new_skills=trace.new_skills,
            new_protocols=trace.new_protocols,
            metrics=self._capture_metrics(),
            rollback_available=True
        )
        
        # 保存到磁盘
        snapshot_path = os.path.join(self.snapshot_dir, f"{version}.json")
        with open(snapshot_path, 'w') as f:
            json.dump(asdict(snapshot), f, indent=2)
        
        # 更新当前版本
        self.current_version = version
        self._save_current_version(version)
        
        # 自动同步到Obsidian
        self._sync_to_obsidian(snapshot)
        
        return snapshot
    
    def rollback(self, target: str) -> bool:
        """回滚到指定版本"""
        if target == "PRE_CYCLE":
            # 回滚到循环前状态
            return self._restore_from_snapshot(self.current_version)
        
        snapshot_path = os.path.join(self.snapshot_dir, f"{target}.json")
        if not os.path.exists(snapshot_path):
            return False
        
        return self._restore_from_snapshot(target)
    
    def diff(self, v1: str, v2: str) -> VersionDiff:
        """版本差异对比"""
        s1 = self._load_snapshot(v1)
        s2 = self._load_snapshot(v2)
        
        return VersionDiff(
            version_from=v1,
            version_to=v2,
            files_added=list(set(s2.files_changed) - set(s1.files_changed)),
            files_removed=list(set(s1.files_changed) - set(s2.files_changed)),
            score_delta=s2.gepa_score.total - s1.gepa_score.total,
            new_skills=list(set(s2.new_skills) - set(s1.new_skills)),
            metric_changes=self._diff_metrics(s1.metrics, s2.metrics)
        )
```

class GoalDrivenIterator:
    """目标驱动定向迭代引擎"""
    
    def __init__(self):
        self.gap_analyzer = GapAnalyzer()
        self.task_generator = TaskGenerator()
        self.validator = GoalValidator()
        self.max_iterations = 3  # 每个缺口最多迭代3轮
    
    def iterate_on_gaps(self) -> IterationReport:
        """扫描缺口并定向迭代"""
        report = IterationReport()
        
        # 1. 缺口识别
        gaps = self.gap_analyzer.scan()  # 基于GEPA矩阵+对标矩阵
        gaps = sorted(gaps, key=lambda g: g.severity, reverse=True)
        report.gaps_found = gaps
        
        # 2. 逐缺口迭代
        for gap in gaps[:5]:  # 每轮最多处理5个缺口
            iteration_count = 0
            
            while iteration_count < self.max_iterations:
                # 2a. 生成定向任务
                tasks = self.task_generator.generate(gap)
                tasks = self._prioritize(tasks)
                
                # 2b. 执行任务（dispatch到对应Agent）
                results = []
                for task in tasks[:3]:
                    agent = self._select_agent(task)
                    result = dispatch_task(
                        agent_name=agent,
                        task=self._build_task(task, gap)
                    )
                    results.append(result)
                
                # 2c. 结果验收
                new_score = self.validator.evaluate(gap, results)
                report.add_iteration(gap.id, iteration_count, new_score)
                
                if new_score >= gap.target_score:
                    report.gaps_closed.append(gap.id)
                    break
                
                iteration_count += 1
            
            if iteration_count >= self.max_iterations:
                report.gaps_unresolved.append(gap.id)
        
        return report

class GapAnalyzer:
    """基于GEPA矩阵+对标矩阵的缺口分析器"""
    
    DIMENSIONS = 28  # 17核心 + 11扩展
    
    def scan(self) -> List[Gap]:
        """扫描所有维度，识别能力缺口"""
        gaps = []
        current_matrix = self.gepa.get_current_matrix()
        target_matrix = self._get_target_matrix()  # 目标矩阵（来自对标基准）
        
        for dim in range(self.DIMENSIONS):
            current = current_matrix[dim]
            target = target_matrix[dim]
            delta = target - current
            
            if delta > 0:  # 存在缺口
                severity = self._calc_severity(delta, dim)
                gaps.append(Gap(
                    id=f"gap_{dim:02d}",
                    dimension=self.dimension_names[dim],
                    current_score=current,
                    target_score=target,
                    delta=delta,
                    severity=severity,  # P0/P1/P2
                    target_score_min=target
                ))
        
        return gaps
```

class ObsidianSyncEngine:
    """Obsidian双向实时同步引擎 v2.0"""
    
    def __init__(self, agent_root: str, obsidian_vault: str):
        self.agent_root = agent_root
        self.obsidian_vault = obsidian_vault
        self.file_watcher = FileWatcher(agent_root)
        self.graph_updater = GraphUpdater(obsidian_vault)
        self.sync_log = SyncLog()
    
    def start(self):
        """启动双向同步监听"""
        self.file_watcher.on_change(self._on_agent_file_changed)
        self.file_watcher.start()
        
        # 启动Obsidian图谱自动更新
        self.graph_updater.start_periodic_update(interval=300)  # 每5分钟
    
    def _on_agent_file_changed(self, event: FileChangeEvent):
        """豆包Agent文件变化 → 自动同步到Obsidian"""
        # 1. 计算目标路径
        rel_path = os.path.relpath(event.path, self.agent_root)
        obs_path = os.path.join(self.obsidian_vault, rel_path)
        
        # 2. 同步文件
        shutil.copy2(event.path, obs_path)
        
        # 3. 更新Obsidian双向链接
        self._update_backlinks(obs_path)
        
        # 4. 更新知识图谱节点
        self.graph_updater.add_or_update_node(rel_path, event.metadata)
        
        # 5. 记录同步日志
        self.sync_log.record(event.path, obs_path, "agent→obsidian")
    
    def query_obsidian(self, query: str) -> List[str]:
        """从Obsidian知识图谱检索内容"""
        # 利用Obsidian的图谱结构进行语义检索
        results = self.graph_updater.search(query)
        return [self._resolve_agent_path(r) for r in results]
    
    def _update_backlinks(self, file_path: str):
        """自动建立双向链接"""
        content = self._read_file(file_path)
        # 提取内部引用
        refs = self._extract_internal_links(content)
        # 在被引用文件中添加反向链接
        for ref in refs:
            ref_path = os.path.join(self.obsidian_vault, ref)
            if os.path.exists(ref_path):
                self._add_backlink(ref_path, file_path)
    
    def sync_retrograde(self, obs_path: str):
        """Obsidian中手动编辑 → 回同步到豆包Agent"""
        rel_path = os.path.relpath(obs_path, self.obsidian_vault)
        agent_path = os.path.join(self.agent_root, rel_path)
        shutil.copy2(obs_path, agent_path)
        self.sync_log.record(obs_path, agent_path, "obsidian→agent")
```

```python
class GraphUpdater:
    """Obsidian知识图谱节点管理器"""
    
    NODE_TYPES = {
        "skill": {"color": "#4f46e5", "icon": "🔧"},
        "architecture": {"color": "#059669", "icon": "🏗️"},
        "iteration": {"color": "#d97706", "icon": "🔄"},
        "benchmark": {"color": "#dc2626", "icon": "📊"},
        "protocol": {"color": "#7c3aed", "icon": "📋"},
        "log": {"color": "#6b7280", "icon": "📝"},
    }
    
    def add_or_update_node(self, rel_path: str, metadata: dict):
        """添加或更新知识图谱节点"""
        node_type = self._classify(rel_path)
        node_config = self.NODE_TYPES.get(node_type, self.NODE_TYPES["log"])
        
        # 生成Obsidian frontmatter
        frontmatter = {
            "tags": [node_type, f"version:{metadata.get('version', 'unknown')}"],
            "created": metadata.get("created", datetime.now().isoformat()),
            "color": node_config["color"],
            "icon": node_config["icon"],
            "aliases": self._generate_aliases(rel_path),
        }
        
        # 写入Obsidian文件的frontmatter
        self._update_frontmatter(rel_path, frontmatter)
    
    def search(self, query: str, top_k: int = 10) -> List[str]:
        """在知识图谱中搜索"""
        # 利用Obsidian的标签系统和图谱结构
        results = []
        for node in self.nodes.values():
            score = self._relevance(query, node)
            if score > 0.5:
                results.append((node.path, score))
        
        results.sort(key=lambda x: x[1], reverse=True)
        return [r[0] for r in results[:top_k]]
```

class DoubaoAgentOrchestratorV2:
    """三引擎统一编排器 v2.0"""
    
    def __init__(self):
        self.reasoning = ClaudeReasoningV6()     # 推理引擎 v6.0
        self.workflow = VisualWorkflowV6()       # 工作流引擎 v6.0
        self.evolution = SelfEvolutionV7()       # 自进化引擎 v7.0

    def execute_full_cycle(self, user_input: str) -> FullCycleResult:
        """执行完整的推理→编排→进化→同步闭环"""

        return FullCycleResult(
