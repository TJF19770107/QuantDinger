### 4.2 节点监控采集器

```python
class NodeMonitor:
    """节点级监控采集器"""
    
    def __init__(self):
        self.metrics_store = MetricsStore()
        self.alert_manager = AlertManager()
    
    def on_node_start(self, node_id: str, node_type: str, agent: str):
        self.metrics_store.init_node(node_id, {
            "node_type": node_type,
            "agent": agent,
            "start_time": time.time(),
            "token_start": self.get_current_token_count()
        })
    
    def on_node_complete(self, node_id: str, result: dict):
        metrics = self.metrics_store.get(node_id)
        metrics["end_time"] = time.time()
        metrics["duration"] = metrics["end_time"] - metrics["start_time"]
        metrics["tokens"] = self.get_current_token_count() - metrics["token_start"]
        metrics["success"] = result.get("success", True)
        
        # 阈值检查
        self._check_thresholds(node_id, metrics)
    
    def _check_thresholds(self, node_id: str, metrics: dict):
        if metrics["duration"] > 60:
            self.alert_manager.fire(node_id, "duration_critical", metrics["duration"])
        elif metrics["duration"] > 30:
            self.alert_manager.fire(node_id, "duration_warning", metrics["duration"])
        
        if metrics["tokens"] > 5000:
            self.alert_manager.fire(node_id, "token_high", metrics["tokens"])
    
    def get_node_health(self, node_id: str) -> NodeHealth:
        metrics = self.metrics_store.get(node_id)
        return NodeHealth(
            status=self._calc_status(metrics),
            performance_score=self._calc_perf(metrics),
            recommendations=self._gen_recommendations(metrics)
        )
```

### 5.3 日志绑定API

```python
class LogBinder:
    """将日志绑定到工作流/节点/操作三层结构"""
    
    def bind_to_workflow(self, workflow_id: str) -> str:
        """创建新的trace_id，绑定到工作流"""
        trace_id = f"wf_{workflow_id}"
        self.active_traces[trace_id] = WorkflowTrace(trace_id)
        return trace_id
    
    def bind_to_node(self, trace_id: str, node_id: str, node_name: str) -> str:
        """创建新的span_id，绑定到节点"""
        span_id = f"{trace_id}.{node_id}"
        self.active_traces[trace_id].add_span(span_id, node_name)
        return span_id
    
    def log_operation(self, span_id: str, operation: str, **kwargs) -> str:
        """记录操作日志，自动绑定到当前span"""
        log_id = f"{span_id}.log_{uuid4().hex[:8]}"
        entry = LogEntry(
            trace_id=self._extract_trace(span_id),
            span_id=span_id,
            log_id=log_id,
            operation=operation,
            **kwargs
        )
        self.log_store.append(entry)
        return log_id
    
    def query_by_trace(self, trace_id: str) -> List[LogEntry]:
        """按trace_id查询全链路日志"""
        return [e for e in self.log_store if e.trace_id == trace_id]
    
    def query_by_node(self, span_id: str) -> List[LogEntry]:
        """按节点查询日志"""
        return [e for e in self.log_store if e.span_id == span_id]
```

### 6.2 多Agent并行分发

```python
class MultiAgentDispatcher:
    """多Agent任务分发器，集成Hermes Swarm DAG拓扑"""
    
    def __init__(self):
        self.agent_router = AgentRouter()
        self.dag_parser = DAGParser()
    
    def dispatch_dag(self, dag: DAG, trace_id: str) -> Dict[str, str]:
        """
        解析DAG拓扑，自动将节点分配到最优Agent，并行派发
        
        Returns:
            {node_id: agent_id} 映射表
        """
        # 1. DAG拓扑分解（Hermes Swarm风格）
        layers = self.dag_parser.topological_layers(dag)
        
        # 2. 逐层并行分发
        assignments = {}
        for layer in layers:
            layer_assignments = {}
            # 同层节点无依赖，并行分发
            for node in layer:
                agent_name = self.agent_router.route(node)
                agent_id = dispatch_task(
                    agent_name=agent_name,
                    task=self._build_task(node, trace_id)
                )
                layer_assignments[node.id] = agent_id
            
            # 等待同层完成再进入下一层
            self._wait_layer(layer_assignments)
            assignments.update(layer_assignments)
        
        return assignments
    
    def _build_task(self, node: DAGNode, trace_id: str) -> str:
        """构建子Agent任务描述，携带trace_id用于日志绑定"""
        return f"""
<overall_goal>
工作流节点执行: {node.name}
trace_id: {trace_id}
</overall_goal>
<current_task>
{node.task_description}
执行完成后将结果写入工作流共享存储。
节点ID: {node.id}
</current_task>
"""
```

```python
class SkillBundleManager:
    """Hermes Swarm风格的Skill Bundles管理"""
    
    def __init__(self):
        self.bundles: Dict[str, SkillBundle] = {}
    
    def bundle_for_workflow(self, workflow_type: str) -> SkillBundle:
        """根据工作流类型加载对应的Skill Bundle"""
        if bundle := self.bundles.get(workflow_type):
            return bundle
        
        # 自动组装Bundle
        bundle = SkillBundle(workflow_type)
        related_skills = self.skill_index.query(workflow_type)
        bundle.add_skills(related_skills)
        self.bundles[workflow_type] = bundle
        return bundle
```

## 八、运行时节点动态增删

### 8.1 动态节点管理

```python
class DynamicNodeManager:
    """运行时动态增删工作流节点"""
    
    def insert_node(self, dag: DAG, after_node_id: str, new_node: DAGNode):
        """在指定节点后插入新节点"""
        # 1. 验证DAG一致性
        if self._would_create_cycle(dag, after_node_id, new_node):
            raise DAGConsistencyError("插入会导致循环依赖")
        
        # 2. 暂停后续节点
        downstream = dag.get_downstream(after_node_id)
        for node in downstream:
            node.pause()
        
        # 3. 插入节点
        dag.insert_after(after_node_id, new_node)
        
        # 4. 恢复后续节点
        for node in downstream:
            node.resume()
    
    def remove_node(self, dag: DAG, node_id: str, bypass: bool = True):
        """移除节点，bypass=True时自动连接前后节点"""
        if bypass:
            upstream = dag.get_upstream(node_id)
            downstream = dag.get_downstream(node_id)
            for u in upstream:
                for d in downstream:
                    dag.add_edge(u, d)
        dag.remove_node(node_id)
```
