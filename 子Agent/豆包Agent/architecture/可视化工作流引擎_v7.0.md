### 3.3 Fallback 降级策略

```python
class ModelRouter:
    """多模型智能路由 with Fallback"""
    
    FALLBACK_CHAIN = [
        Tier.OPUS_4_8,      # 首选
        Tier.SONNET_4,      # 次选
        Tier.DEEPSEEK_V3_1, # 兜底
    ]
    
    async def route(self, node: WorkflowNode) -> ModelDecision:
        """为工作流节点选择最优模型"""
        complexity = self._assess_complexity(node)
        security_required = node.requires_mcp_sandbox
        
        if security_required:
            # 安全节点: 必须过MCP沙箱
            model = Tier.MSB_SECURE
        elif complexity > 0.7:
            model = Tier.OPUS_4_8
        elif complexity > 0.4:
            model = Tier.SONNET_4
        else:
            model = Tier.DEEPSEEK_V3_1
        
        # 带重试的fallback
        decision = await self._try_execute(node, model)
        if decision.failed:
            for fallback_model in self.FALLBACK_CHAIN:
                decision = await self._try_execute(node, fallback_model)
                if not decision.failed:
                    break
        
        return decision
```

### 4.1 节点持久化架构

```
传统工作流节点:
  ┌─────────┐     崩溃后
  │ Node 1  │ → 全部重跑 Node 1 → Node 2 → Node 3

  │ Node 3  │

Durable 持久化节点:
  ┌─────────┐     崩溃后
```

## 六、工作流版本快照与回滚

### 6.1 版本快照机制

```python
class WorkflowSnapshotManager:
    """工作流版本快照管理器"""
    
    SNAPSHOT_INTERVAL = 5  # 每5次变更自动快照
    
    def create_snapshot(self, workflow_id: str) -> Snapshot:
        """创建完整工作流快照"""
        return Snapshot(
            id=f"snap_{workflow_id}_{timestamp()}",
            workflow_id=workflow_id,
            node_graph=self._serialize_dag(),        # DAG拓扑
            node_states=self._serialize_states(),     # 所有节点状态
            model_routing=self._serialize_routing(),  # 模型路由配置
            security_policies=self._serialize_policies(), # 安全策略
            metadata={
                'version': self.current_version,
                'total_nodes': len(self.nodes),
                'active_models': self.active_models,
            }
        )
    
    def rollback_to(self, snapshot_id: str) -> bool:
        """精确回滚到指定快照"""
        snapshot = self._load_snapshot(snapshot_id)
        self._restore_dag(snapshot.node_graph)
        self._restore_states(snapshot.node_states)
        self._restore_routing(snapshot.model_routing)
        return True
```
