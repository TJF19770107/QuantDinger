```python
"""
豆包Agent v5.0 核心执行循环
对标: Codex Agent Loop + Hermes Learning Loop + Ouroboros Consciousness
"""
import asyncio
import sqlite3
from dataclasses import dataclass
from typing import Any, Callable

@dataclass
class AgentState:
    """Agent状态追踪"""
    task_id: str
    step: int = 0
    max_steps: int = 40
    context: list[str] = None  # 对话历史
    tool_results: list[dict] = None
    skills_loaded: list[str] = None  # 已加载技能
    reflection_notes: str = ""

class DoubaoAgentCore:
    """豆包Agent核心引擎"""
    
    def __init__(self):
        self.state = AgentState()
        self.memory = TripleLayerMemory()  # 三层持久记忆
        self.mailbox = AgentMailbox()       # Agent间通信
        self.tools = ToolRegistry()          # 工具注册表
        self.skills = SkillForge()          # 技能锻造厂
        self.sandbox = SafetySandbox()       # 安全沙箱
        self.constitution = Constitution()   # 核心宪法
    
    async def agent_loop(self, task: str) -> str:
        """主执行循环: 对标Codex Agent Loop"""
        self.state.context.append(f"[User] {task}")
        
        while self.state.step < self.state.max_steps:
            self.state.step += 1
            
            # 1. 观察 (Observe)
            observation = await self.observe()
            
            # 2. 思考 (Think)
            decision = await self.think(observation)
            
            # 3. 行动 (Act) - 含Smart Approvals
            if decision["action"] == "respond":
                return decision["response"]
            
            result = await self.act(decision, observation)
            
            # 4. 反馈 (Feedback)
            self.state.context.append(result)
            
            # 5. 反思 (Reflect) - Hermes闭环
            if self._is_complex_task():
                await self.reflect_and_crystallize()
    
    async def observe(self) -> dict:
        """观察当前状态"""
        return {
            "context": self.state.context[-20:],
            "tools_available": self.tools.list(),
            "skills_loaded": self.state.skills_loaded,
            "sandbox_status": self.sandbox.status(),
        }
    
    async def think(self, obs: dict) -> dict:
        """思考决策: 模型路由 + 工具选择"""
        model = self._route_model(obs)  # 自动选择最优模型
        prompt = self._build_structured_prompt(obs)
        response = await model.generate(prompt)
        return self._parse_decision(response)
    
    async def act(self, decision: dict, obs: dict) -> str:
        """执行行动: 含Smart Approvals安全检查"""
        # Smart Approvals: 守护代理审查
        approval = await self.sandbox.approve(decision)
        if approval == "BLOCK":
            return "[BLOCKED] 操作被安全策略拦截"
        elif approval == "ESCALATE":
            return await self._escalate_to_user(decision)
        
        # 隔离执行
        return await self.sandbox.execute(decision["tool"], decision["params"])
    
    def _route_model(self, obs: dict) -> str:
        """智能模型路由: 对标OpenCode + Gemini Flash"""
        if self._is_coding_task(obs):
            return "deepseek-v4"  # 编码最优
        elif self._is_multimodal_task(obs):
            return "gemini-3.5-flash"  # 多模态+性价比
        elif self._is_deep_reasoning(obs):
            return "hunyuan3"  # 深度推理
        else:
            return "qwen2.5-7b"  # 本地轻量兜底
    
    async def reflect_and_crystallize(self):
        """反思与结晶: 对标Hermes闭环学习"""
        # 分析任务执行轨迹
        trajectory = self.state.context
        skill_candidate = await self.skills.extract(trajectory)
        
        if skill_candidate.confidence > 0.7:
            # 沉淀为可复用Skill
            self.skills.save(skill_candidate)
            self.memory.store_longterm(skill_candidate)


class TripleLayerMemory:
    """三层持久记忆: 对标Hermes"""
    
    def __init__(self):
        self.short_term = []           # L1: 会话上下文
        self.medium_term = sqlite3.connect(":memory:")  # L2: FTS5
        self.long_term = SkillForge()  # L3: 技能文件
    
    def store_longterm(self, skill):
        """长期记忆: 技能文件沉淀"""
        self.long_term.save(skill)
        self._sync_to_fs()  # 同步到文件系统
    
    def recall(self, query: str) -> list:
        """跨层检索"""
        results = []
        results += self._search_short(query)
        results += self._search_fts5(query)
        results += self._search_skills(query)
        return results


class AgentMailbox:
    """Agent间消息队列: 对标Claude Agent Teams"""
    
    def __init__(self):
        self.queues: dict[str, asyncio.Queue] = {}
    
    async def send(self, to_agent: str, message: dict):
        """发送消息到指定Agent"""
        if to_agent not in self.queues:
            self.queues[to_agent] = asyncio.Queue()
        await self.queues[to_agent].put(message)
    
    async def receive(self, agent_id: str, timeout: float = 30):
        """接收消息"""
        if agent_id not in self.queues:
            self.queues[agent_id] = asyncio.Queue()
        return await asyncio.wait_for(
            self.queues[agent_id].get(), timeout=timeout
        )


class SafetySandbox:
    """安全沙箱: 对标Codex Docker + Seatbelt"""
    
    def __init__(self):
        self.blocked_commands = {"rm -rf", "format", "del /f", "reg delete"}
        self.constitution = Constitution()
    
    async def approve(self, decision: dict) -> str:
        """Smart Approvals: 对标Codex守护代理"""
        # L1: 硬编码安全规则
        if self._matches_blocked(decision):
            return "BLOCK"
        
        # L2: LLM安全代理审查
        if self._is_destructive(decision):
            safety_check = await self._llm_safety_review(decision)
            if not safety_check:
                return "BLOCK"
        
        # L3: 宪法原则审查
        if not self.constitution.validate(decision):
            return "ESCALATE"
        
        return "APPROVE"
    
    async def execute(self, tool: str, params: dict) -> str:
        """隔离沙箱执行"""
        # 预执行模拟 (dry-run)
        simulation = self._dry_run(tool, params)
        if simulation.risks:
            return f"[RISK] {simulation.risks}"
        
        # 快照备份 (破坏性操作前)
        if self._is_destructive({"tool": tool}):
            self._create_snapshot()
        
        # 实际执行
        return await self._safe_exec(tool, params)


class SkillForge:
    """技能锻造厂: 对标Hermes Skill Auto-Generation"""
    
    def __init__(self, skills_dir: str):
        self.skills_dir = skills_dir
    
    async def extract(self, trajectory: list) -> "SkillCandidate":
        """从任务轨迹中提取可复用技能"""
        # 分析执行步骤
        steps = self._identify_steps(trajectory)
        # 识别可复用模式
        pattern = self._find_pattern(steps)
        # 评估置信度
        confidence = self._evaluate_reusability(pattern)
        return SkillCandidate(pattern, confidence)
    
    def save(self, skill: "SkillCandidate"):
        """保存为SKILL.md"""
        skill_path = f"{self.skills_dir}/{skill.name}.md"
        with open(skill_path, "w") as f:
            f.write(skill.to_markdown())
```
