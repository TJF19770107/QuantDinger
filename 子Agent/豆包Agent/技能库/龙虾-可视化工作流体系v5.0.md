### 3.2 DAG拓扑排序算法

```python
# dag_builder.py
from collections import deque
from typing import List, Dict, Set

class DAGBuilder:
    """
    DAG拓扑构建器
    对标 Sim Studio AI DAGBuilder
    """
    
    def __init__(self, workflow_json: dict):
        self.nodes = workflow_json.get("nodes", [])
        self.edges = workflow_json.get("edges", [])
        self.adjacency = self._build_adjacency()
        self.in_degree = self._compute_in_degrees()
    
    def topological_sort(self) -> List[str]:
        """
        Kahn算法拓扑排序
        返回: 可并行执行的层级列表
        """
        queue = deque([
            node_id for node_id, deg in self.in_degree.items()
            if deg == 0
        ])
        
        levels = []  # 每层可并行执行的节点
        while queue:
            level = []
            for _ in range(len(queue)):
                node = queue.popleft()
                level.append(node)
                for neighbor in self.adjacency.get(node, []):
                    self.in_degree[neighbor] -= 1
                    if self.in_degree[neighbor] == 0:
                        queue.append(neighbor)
            levels.append(level)
        
        return levels  # [[并行层1], [并行层2], ...]
    
    def detect_cycles(self) -> bool:
        """环路检测"""
        visited = set()
        rec_stack = set()
        
        def dfs(node):
            visited.add(node)
            rec_stack.add(node)
            for neighbor in self.adjacency.get(node, []):
                if neighbor not in visited:
                    if dfs(neighbor):
                        return True
                elif neighbor in rec_stack:
                    return True
            rec_stack.remove(node)
            return False
        
        for node in self.nodes:
            if node["id"] not in visited:
                if dfs(node["id"]):
                    return True
        return False
```

### 4.2 Handler注册与扩展

```python
# block_registry.py
from abc import ABC, abstractmethod
from typing import Any, Dict

class BlockHandler(ABC):
    """Block Handler 基类"""
    
    @abstractmethod
    async def execute(self, inputs: Dict[str, Any], context: Dict) -> Dict[str, Any]:
        """执行Block逻辑"""
        pass
    
    @abstractmethod
    def validate_inputs(self, inputs: Dict[str, Any]) -> bool:
        """验证输入格式"""
        pass

class BlockRegistry:
    """Handler注册中心"""
    
    _handlers: Dict[str, type] = {}
    
    @classmethod
    def register(cls, handler_type: str, handler_class: type):
        cls._handlers[handler_type] = handler_class
    
    @classmethod
    def get(cls, handler_type: str) -> type:
        return cls._handlers.get(handler_type, GenericHandler)

# 注册示例
BlockRegistry.register("agent", AgentHandler)
BlockRegistry.register("condition", ConditionHandler)
BlockRegistry.register("router", RouterHandler)
```

### 5.2 ParallelOrchestrator（并行编排器）

```python
# parallel_orchestrator.py
class ParallelOrchestrator:
    """
    并行编排器
    对标 Sim Studio AI ParallelOrchestrator
    """
    
    async def execute_parallel(
        self, 
        branches: List[WorkflowBranch],
        max_parallel: int = 16
    ) -> Dict[str, Any]:
        """
        并行执行多个分支
        自动管理信号量和资源限制
        """
        semaphore = asyncio.Semaphore(max_parallel)
        
        async def run_branch(branch):
            async with semaphore:
                return await self._execute_single_branch(branch)
        
        tasks = [run_branch(b) for b in branches]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        return self._merge_results(results)
```

## 八、断路器熔断机制 v2.0

### 8.1 三态断路器

```
         ┌─────────┐
         │ CLOSED  │ (正常)
         └────┬────┘
              │ 失败次数 ≥ 阈值
              ▼
         ┌─────────┐
         │  OPEN   │ (熔断)
         └────┬────┘
              │ 超时后
              ▼
         ┌─────────────┐
         │ HALF_OPEN   │ (试探)
         └────┬────────┘
              │
     ┌───────┴───────┐
     ▼               ▼
  成功→CLOSED    失败→OPEN
```

### 8.2 熔断配置

| 参数 | 默认值 | 说明 |
|------|--------|------|
| failure_threshold | 5 | 连续失败次数阈值 |
| timeout_seconds | 60 | OPEN→HALF_OPEN等待时间 |
| half_open_max_requests | 3 | 试探请求数 |
| success_threshold | 2 | HALF_OPEN→CLOSED成功阈值 |
| backoff_multiplier | 2 | 指数退避倍数 |
