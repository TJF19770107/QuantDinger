# skill_forge.py - 技能自动锻造器
class SkillForge:
    def extract_patterns(self, iteration_logs):
        """从迭代日志中提取可复用模式"""
        patterns = []
        for log in iteration_logs:
            # 识别"执行了X操作"→"产生了Y结果"
            actions = self.parse_actions(log)
            for action in actions:
                if action.is_reusable():
                    patterns.append(action.to_skill_template())
        return patterns
    
    def generate_skill(self, pattern):
        """根据模式生成完整技能包"""
        skill = {
            "name": pattern.name,
            "triggers": pattern.extract_triggers(),
            "steps": pattern.extract_steps(),
            "code": pattern.generate_code(),
            "safety": self.safety_review(pattern)
        }
        self.save_to_skill_library(skill)
        self.register_triggers(skill)
        return skill
```

        """执行脚本（安全审查后）"""
        if self.safety_check(script_path):

#### 5.4 记忆压缩策略（对标Letta虚拟内存）

# memory_os.py - 记忆操作系统
class MemoryOS:
    def __init__(self):
        self.session_memory = {}   # Layer 2
        self.long_term_db = None   # Layer 3 (SQLite)
        
    def auto_load(self):
        """自动加载：最新快照 + 相关长期记忆"""
        snapshot = self.load_latest_snapshot()
        self.session_memory = snapshot["session"]
        relevant = self.query_long_term(snapshot["context_keys"])
        return {**self.session_memory, "long_term": relevant}
    
    def auto_compress(self):
        """自动压缩：超阈值记忆→LLM摘要"""
        for memory_id, entry in self.long_term_db.items():
            if entry["token_count"] > self.COMPRESS_THRESHOLD:
                summary = self.llm_summarize(entry["content"])
                entry["content"] = summary
                entry["compressed"] = True
    
    def auto_decay(self):
        """自动衰减：长期未用记忆降低权重"""
        now = time.time()
        for memory_id, entry in self.long_term_db.items():
            days_since_access = (now - entry["last_access"]) / 86400
            if days_since_access > 30:
                entry["weight"] *= 0.5  # 权重减半
```

# safe_guard.py - 安全护栏系统
class SafeGuard:
    def __init__(self, checkpoint_dir):
        self.checkpoint_dir = Path(checkpoint_dir)
        self.stable = self.load_stable_checkpoint()
        
    def pre_exec_review(self, code_diff):
        """执行前审查"""
        dangerous = ["eval(", "exec(", "os.system(", "subprocess.call", 
                      "rm -rf", "del /f", "format "]
        for line in code_diff["changes"]:
            for danger in dangerous:
                if danger in line:
                    return {"verdict": "UNSAFE", "reason": f"Contains {danger}"}
        return {"verdict": "SAFE"}
    
    def runtime_monitor(self, process):
        """运行时监控"""
        if process.cpu_percent > 90:
            process.terminate()
            return {"status": "TERMINATED", "reason": "CPU spike"}
        if process.runtime > process.expected_runtime * 2:
            process.terminate()
            return {"status": "TIMEOUT", "reason": "Exceeded 200% time budget"}
    
    def auto_rollback(self):
        """自动回滚到上一稳定版本"""
        stable = self.load_stable_checkpoint()
        for file_path, expected_md5 in stable["snapshot"].items():
            current_md5 = self.hash_file(file_path)
            if current_md5 != expected_md5:
                self.restore_from_checkpoint(file_path, stable)
        self.log_rollback()
