## DataviewJS 动态策略仪表盘

```dataviewjs
const strategies = dv.pages('"quant_data/strategies"');

// 统计各状态数量
const statusCount = {};
strategies.forEach(s => {
    const status = s.status || "未分类";
    statusCount[status] = (statusCount[status] || 0) + 1;
});

dv.header(3, "策略状态分布");
const statusRows = Object.entries(statusCount)
    .sort((a, b) => b[1] - a[1])
    .map(([status, count]) => [status, count]);
dv.table(["状态", "策略数"], statusRows);

// 标的分布
const assetMap = {};
strategies.forEach(s => {
    const asset = s.asset || "未指定";
    assetMap[asset] = (assetMap[asset] || 0) + 1;
});

dv.header(3, "标的覆盖");
const assetRows = Object.entries(assetMap)
    .sort((a, b) => b[1] - a[1])
    .map(([asset, count]) => [asset, count]);
dv.table(["标的", "策略数"], assetRows);

dv.paragraph(`**策略总数**：${strategies.length} | **状态类型**：${Object.keys(statusCount).length} | **覆盖标的**：${Object.keys(assetMap).length}`);
```
