## DataviewJS 动态交易仪表盘

```dataviewjs
const reviews = dv.pages('"共享知识库" or "quant_data"')
    .where(p => p.type === "review" || p.file_type === "验收报告");

const reviewList = reviews.sort(p => p.date || p.file.ctime, 'desc');

dv.header(3, "复盘记录总览");
dv.paragraph(`**记录总数**：${reviewList.length}`);

// 按年月分组统计
const monthCount = {};
reviewList.forEach(r => {
    const d = r.date || r.file.ctime;
    if (d) {
        const monthKey = d.toString().substring(0, 7); // YYYY-MM
        monthCount[monthKey] = (monthCount[monthKey] || 0) + 1;
    }
});

dv.header(4, "月度复盘统计");
const monthRows = Object.entries(monthCount)
    .sort((a, b) => b[0].localeCompare(a[0]))
    .map(([month, count]) => [month, count]);
dv.table(["月份", "复盘数"], monthRows);
```
