## DataviewJS 动态统计仪表盘

```dataviewjs
// 资产总览动态统计
const pages = dv.pages('"quant_data"').where(p => p.file_type);

// 按类型统计
const typeCount = {};
pages.forEach(p => {
    const type = p.file_type;
    if (type) {
        typeCount[type] = (typeCount[type] || 0) + 1;
    }
});

// 按标签统计
const tagCount = {};
pages.forEach(p => {
    if (p.tags) {
        if (Array.isArray(p.tags)) {
            p.tags.forEach(t => { tagCount[t] = (tagCount[t] || 0) + 1; });
        }
    }
});

dv.header(3, "统计概览");
dv.paragraph(`**总文件数**：${pages.length} | **文件类型数**：${Object.keys(typeCount).length} | **标签总数**：${Object.keys(tagCount).length}`);

// 类型分布表
dv.header(4, "文件类型分布");
const typeRows = Object.entries(typeCount)
    .sort((a, b) => b[1] - a[1])
    .map(([type, count]) => [type, count]);
dv.table(["文件类型", "数量"], typeRows);

// 热门标签 Top 10
dv.header(4, "热门标签 Top 10");
const tagRows = Object.entries(tagCount)
    .sort((a, b) => b[1] - a[1])
    .slice(0, 10)
    .map(([tag, count]) => [tag, count]);
dv.table(["标签", "出现次数"], tagRows);
```
