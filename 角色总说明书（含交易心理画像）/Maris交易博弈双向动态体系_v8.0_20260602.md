### 10.1 核心监控脚本架构

```python

class MarisMonitor:
    """
    Maris博弈体系 v8.0 实时监控引擎
    功能：五重熔断 / 主力动向识别 / 预警推送 / 自动拦截
    """

    def check_circuit_breakers(self):
        """五重熔断检查"""
        breakers = {
            "CB1_ETF": self.check_etf_outflow(),       # ETF日流出>$300M
            "CB2_RECOVERY": self.check_liquidation_recovery(), # 爆仓恢复Day1-7
            "CB3_FEAR": self.check_fear_index(),       # 恐惧指数≥tier_10
            "CB4_VOLATILITY": self.check_volatility(), # 波动率超标
            "CB5_WAR_MACRO": self.check_war_macro(),   # 战争/宏观经济冲击
        }

        active = [k for k, v in breakers.items() if v]
        if active:
            return {
                "status": "LOCKED",
                "active_breakers": active,
                "auto_block": True,
                "earliest_trade": self.get_earliest_trade_date(),
                "message": f"五重熔断中 {len(active)}/5 — 全账户锁定"
            }
        return {"status": "CLEAR", "active_breakers": []}

    def detect_main_force(self, symbol):
        """五阶段主力动向识别"""
        # 获取实时数据
        ohlcv = self.get_ohlcv(symbol)
        volume = self.get_volume(symbol)
        large_orders = self.get_large_orders(symbol)
        sentiment = self.get_sentiment(symbol)

        # 五阶段判定
        stage = self.classify_stage(ohlcv, volume, large_orders, sentiment)
        confidence = self.get_confidence(stage)

        return {
            "symbol": symbol,
            "stage": stage,        # accumulation/washout/pump/distribution/oscillation
            "confidence": confidence,
            "recommended_action": self.map_stage_to_action(stage, confidence)
        }

    def map_stage_to_action(self, stage, confidence):
        """主力阶段 → 个人操作映射"""
        mapping = {
            "accumulation": ("MARK_FOCUS", "P2", 0),
            "washout":      ("REVERSE_POSITION", "P1", 0.20),
            "pump":         ("PYRAMID_ADD", "P0", 0.50),
            "oscillation":  ("REDUCE_LOCK", "P1", -0.30),
            "distribution": ("LIQUIDATE_ALL", "P0", -1.00),
        }
        return mapping.get(stage, ("WAIT", "P2", 0))
```
